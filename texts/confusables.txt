# Blank lines and lines starting with `#` are skipped.
#
#     cargo run --release -p dataset-gen -- \
#         --body-fonts fonts/ja-train --deco-fonts fonts/ja-train \
#         --text-file texts/confusables.txt --count 20000 --out dataset/confusables
#     uv run python -m glyph_gnn.eval --checkpoint <ckpt> --data dataset/confusables
#
# Every line is rendered the same number of times across fonts and
# augmentations, so the per-pattern accuracies in the eval breakdown are
# directly comparable. Each merge case is paired with the single character it

# model fixes `ll` by splitting harder, `m` and `w` start splitting too.
